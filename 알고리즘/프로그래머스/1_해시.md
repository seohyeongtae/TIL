```java
		}
		return;
	}
```

> Comparator 를 이용한 정렬 후 dfs
>
> 그래도 1번 시간초과 뜬다 -> 성능(효율성) 개선은 되었음.

```java
import java.util.Arrays;
import java.util.Comparator;
import java.util.HashSet;

class Solution {
	static HashSet hm = new HashSet<String>();
	static int answer = 0;

    public int solution(String[][] clothes) {
    	Arrays.sort(clothes, new Comparator<String []>()
			{
				@Override
				public int compare(String[] o1, String[] o2) {
						return o1[1].compareTo(o2[1]);
				}
			});
		dfs(clothes, 0);
		//System.out.println(answer);
        return answer;
    }

	private void dfs(String[][] clothes, int i) {
		
		if (i == clothes.length)
			return;
		for (int j = i; j < clothes.length; j++)
		{
			if (!(hm.contains(clothes[j][1])))
			{
				hm.add(clothes[j][1]);
				answer++;
				dfs(clothes, j + 1);
//				System.out.println(clothes[i][1]);
				hm.remove(clothes[j][1]);
			}
		}
		return;
	}	
}
```

> 다른사람 정답 코드 (수학공식 활용)
>
> (하의 2개+안입은 경우) * (상의 3개 + 안입은 경우) = 3 * 4 = 12
> 그리고 전체에서 -1 (전부다 안입는 경우의 수를 뺀다)  = 11

```java
import java.util.*;

class Solution {
	public int solution(String[][] clothes) {
		int answer = clothes.length;
		HashMap<String, Integer> map = new HashMap<>();
	        for(int i=0; i<clothes.length; i++) {
	            map.put(clothes[i][1], map.getOrDefault(clothes[i][1], 0) + 1);
	        }
	
	    //key가 2개 이상이면
	        if(map.size() > 1) {
	            int count = 1;
	            for(String key : map.keySet()){
	                count *= map.get(key) + 1;
	            }
	            answer = count - 1;
	        }
	        System.out.println(answer);
	        return answer;
	    }
	}
```

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map.Entry;

class Solution {
	
    public int[] solution(String[] genres, int[] plays) {
        int[] answer = null;
        HashMap hm = new HashMap<String, Integer>();
        for (int i = 0; i < genres.length; i++)
        	hm.put(genres[i], ((int)(hm.getOrDefault(genres[i], 0)))  +  plays[i]);
        List<Entry<String, Integer>> list_entries = new ArrayList<Entry<String, Integer>>(hm.entrySet());
    	Collections.sort(list_entries, new Comparator<Entry<String, Integer>>() {
			// compare로 값을 비교
			public int compare(Entry<String, Integer> obj1, Entry<String, Integer> obj2)
			{
				// 내림 차순으로 정렬
				return obj2.getValue().compareTo(obj1.getValue());
			}
		});
    	
    	String ger[] = new String[list_entries.size()];
    	int cnt = 0;
    	for(Entry<String, Integer> entry : list_entries) 
    	{
    		ger[cnt] = entry.getKey();
    		cnt++;
    	}
    	
//    	for (int i = 0; i < ger.length; i++)
//    		System.out.println(ger[i]);

    	int result[][] = new int[ger.length][2];  // i 번째 1번째, 2번째
    	int result_n[][] = new int[ger.length][2]; // plays 수
    	for (int i = 0; i < genres.length; i++)
    	{
    		for (int j = 0; j < ger.length; j++)
    		{
    			if (ger[j].equals(genres[i]))
    			{
    				if (result_n[j][0] == 0)
    				{
    					result[j][0] = i;
    					result_n[j][0] = plays[i];
    				}
    				else
    				{
    					if (plays[i] > result_n[j][0]) 
    					{
    						result[j][1] = result[j][0];
        					result_n[j][1] = result_n[j][0];
        					result[j][0] = i;
        					result_n[j][0] = plays[i];
    					}
    					else if (plays[i] > result_n[j][1])
    					{
    						result[j][1] = i;
    						result_n[j][1] = plays[i];
    					}
    				}
    			}
    		}
    	}
    	ArrayList<Integer> al = new ArrayList<Integer>();
    	for (int i = 0; i < ger.length; i++)
    	{
    		if (result_n[i][1] != 0)
    		{
    			al.add(result[i][0]);
    			al.add(result[i][1]);
    		}
    		else al.add(result[i][0]);
    	}
    	answer = new int[al.size()];
    	for (int i = 0; i < al.size(); i++)
    		answer[i] = al.get(i);
//    	for (int i = 0; i < answer.length; i++)
//    		System.out.println(answer[i]);
        return answer;
    	}
}
```

### 스택 (10828번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			String instruction = st.nextToken();
			if (instruction.equals("push"))
			{
				int num = Integer.parseInt(st.nextToken());
				stack.push(num);
			}
			if (instruction.equals("top"))
			{
				if (stack.size() != 0)
				{	
					System.out.println(stack.lastElement());					
				} else
					System.out.println("-1");
			}
			if (instruction.equals("size"))
			{
				System.out.println(stack.size());
			}
			if (instruction.equals("empty"))
			{
				if (stack.empty())
				{
					System.out.println("1");					
				} else
					System.out.println("0");
			}
			if (instruction.equals("pop"))
			{
				if (stack.empty())
				{
					System.out.println("-1");					
				} else
				{
					System.out.println(stack.lastElement());
					stack.pop();
				}
			}
		}
	} // main end
} // class end
```

```java
			int num = Integer.parseInt(st.nextToken());
			if (num == 0)
			{
				stack.pop();
			}else
			{
				stack.push(num);
			}
		}
		for (int i = 0; i < stack.size(); i++)
		{
			result += stack.get(i);
		}
		System.out.println(result);
```

### 괄호 (9012번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			String ps = st.nextToken();
			for (int j = 0; j < ps.length(); j++)
			{
				if (ps.charAt(j) == '(')
				{
					stack.push(j);
				}
				else if (ps.charAt(j) == ')')
				{
					if (stack.empty())
					{
						System.out.println("NO");
						break;
					}
					else
					{
						stack.pop();
					}
				}
				if (j == ps.length() - 1)
				{
					if (!(stack.empty()))
						System.out.println("NO");
					else 
						System.out.println("YES");
				}
			}
			stack.clear();
		}
	} // main end
} // class end
```

### 균형잡힌 세상 (4949번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	
		while (true)
		{
			Stack<String> stack = new Stack<String>();

			String temp = br.readLine();
			if (temp.charAt(0) == '.')
			{
				break;				
			}
			for (int i = 0; i < temp.length(); i++)
			{
				if (temp.charAt(i) == '(')
				{
					stack.push("(");
				}
				else if (temp.charAt(i) == ')')
				{
					if (stack.empty() || !(stack.lastElement().equals("(")))
						{
							System.out.println("no");
							break;
						}
						else
						{
							stack.pop();
						}
				}
				if (temp.charAt(i) == '[')
				{
					stack.push("[");
				}
				else if (temp.charAt(i) == ']')
				{
					if (stack.empty() || !(stack.lastElement().equals("[")))
						{
							System.out.println("no");
							break;
						}
						else
						{
							stack.pop();
						}
				}
				if (i == temp.length() - 1)
					{
						if (!(stack.empty()))
							System.out.println("no");
						else 
							System.out.println("yes");
					}
			}
			stack.clear();
		} // while 종료
	} // main end
} // class end
```

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int N = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		StringBuilder sb = new StringBuilder();
		st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int input = 0;
		for (int i = 1; i <= n; i++)
		{
			stack.add(i);
			input = i;
			sb.append("+" + "\n");
		}
		stack.pop();
		sb.append("-" + "\n");
		for (int i = 1; i < N; i++)
		{
			st = new StringTokenizer(br.readLine());
			n = Integer.parseInt(st.nextToken());
			int temp = input;
			for (int j = input + 1; j <= n; j++)
			{
				sb.append("+" + "\n");
				stack.add(j);
				temp = j;
			}
			input = temp;
			if (stack.lastElement() == n)
			{
				sb.append("-" + "\n");
				stack.pop();
			}
			else
			{
				System.out.println("NO");
				return;
			}
		}
		System.out.println(sb);
	} // main end
} // class end
```

> 스택 2개 사용한 코드

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int N = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		Stack<Integer> stack2 = new Stack<Integer>();
		StringBuilder sb = new StringBuilder();
		st = new StringTokenizer(br.readLine());
		int result[] = new int[N];
		for (int i = 0; i < N; i++)
		{
			stack.add(Integer.parseInt(st.nextToken()));
		}
		int max = stack.peek();
		int real_max = stack.peek();
		result[N - 1] = -1;
		stack2.add(stack.pop());
		for (int i = 1; i < N; i++)
		{
			if (stack.peek() >= real_max)
			{
				real_max = stack.peek();
				result[N - i - 1] = -1;
				stack2.add(stack.pop());
				
			}
			else if (stack.peek() < stack2.peek())
			{
				max = stack2.peek();
				stack2.add(stack.pop());
				result[N - i - 1] = max;
			}
			else if (stack.peek() >= stack2.peek())
			{
				while (stack.peek() >= stack2.peek())
				{
					stack2.pop();
				}

				if (stack.peek() < stack2.peek())
				{
					max = stack2.peek();
					stack2.add(stack.peek());
					result[N - i - 1] = max;
				}
				else
				{
					max = stack2.peek();
					result[N - i - 1] = max;
				}
				stack2.add(stack.peek());
				stack.pop();
			}
		}
		for (int i = 0; i < N; i++)
			sb.append(result[i]).append(' ');
		System.out.println(sb);
	} // main end
} // class end
```

>다른 사람 코드 (스택 1개 사용)
>
>스택에 인덱스 값을 집어 넣음.

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;
import java.util.Stack;
 
public class Main {
 
	public static void main(String[] args) throws IOException {
		
		
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		Stack<Integer> stack = new Stack<Integer>();
		
		int N = Integer.parseInt(br.readLine());
		int[] seq = new int[N];
		
		
		StringTokenizer st = new StringTokenizer(br.readLine(), " ");
		
		for(int i = 0; i < N; i++) {
			seq[i] = Integer.parseInt(st.nextToken());
		}
 
 
		for(int i = 0; i < N; i++) {
			
			/*
			 * 스택이 비어있지 않으면서 
			 * 현재 원소가 스택의 맨 위 원소가 가리키는 원소보다 큰 경우
			 * 해당 조건을 만족할 때 까지 stack의 원소를 pop하면서
			 * 해당 인덱스의 값을 현재 원소로 바꿔준다. 
			 */
			while(!stack.isEmpty() && seq[stack.peek()] < seq[i]) {
				seq[stack.pop()] = seq[i];
			}
			
			stack.push(i);
		}
		
		/*
		 * 스택의 모든 원소를 pop하면서 해당 인덱스의 value를
		 * -1로 초기화한다.
		 */
		while(!stack.isEmpty()) {
			seq[stack.pop()] = -1;
		}
		
		StringBuilder sb = new StringBuilder();
		for(int i = 0; i < N; i++) {
			sb.append(seq[i]).append(' ');
		}
		
		System.out.println(sb);
	}
}
```

> 스택 대신 배열 사용 코드

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;
 
public class Main {
	public static void main(String[] args) throws IOException {
 
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
 
		int N = Integer.parseInt(br.readLine());
		int[] seq = new int[N];
		int[] stack = new int[N];	// 스택처럼 쓸 배열 
		int top = -1;	// 스택의 맨 위 원소를 가리키는 변수
		
		StringTokenizer st = new StringTokenizer(br.readLine(), " ");
		for (int i = 0; i < N; i++) {
			seq[i] = Integer.parseInt(st.nextToken());
		}
		
 
		for (int i = 0; i < N; i++) {
			/*
			 * 스택이 비어있지 않으면서 (= top 이 -1이 아닐 경우) 
			 * 현재 원소가 스택의 맨 위 원소가 가리키는 원소보다 큰 경우
			 * 해당 조건을 만족할 때 까지 stack의 원소를 pop하면서
			 * 해당 인덱스의 값을 현재 원소로 바꿔준다. 
			 */
			while (top != - 1 && seq[stack[top]] < seq[i]) {
				seq[stack[top]] = seq[i];
				top--;
			}
			top++;
			stack[top] = i;
		}
 
		/*
		 * 스택의 모든 원소를 pop하면서 해당 인덱스의 value를
		 * -1로 초기화한다.
		 */
		for(int i = top; i >= 0; i--) {
			seq[stack[i]] = -1;
		}
 
		StringBuilder sb = new StringBuilder();
		for (int v : seq) {
			sb.append(v).append(' ');
		}
 
		System.out.println(sb);
	}
}
```

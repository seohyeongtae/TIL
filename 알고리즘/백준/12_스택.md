### 스택 (10828번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			String instruction = st.nextToken();
			if (instruction.equals("push"))
			{
				int num = Integer.parseInt(st.nextToken());
				stack.push(num);
			}
			if (instruction.equals("top"))
			{
				if (stack.size() != 0)
				{	
					System.out.println(stack.lastElement());					
				} else
					System.out.println("-1");
			}
			if (instruction.equals("size"))
			{
				System.out.println(stack.size());
			}
			if (instruction.equals("empty"))
			{
				if (stack.empty())
				{
					System.out.println("1");					
				} else
					System.out.println("0");
			}
			if (instruction.equals("pop"))
			{
				if (stack.empty())
				{
					System.out.println("-1");					
				} else
				{
					System.out.println(stack.lastElement());
					stack.pop();
				}
			}
		}
	} // main end
} // class end
```

		Stack<Integer> stack = new Stack<Integer>();
		int result = 0;
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			int num = Integer.parseInt(st.nextToken());
			if (num == 0)
			{
				stack.pop();
			}else
			{
				stack.push(num);
			}
		}
		for (int i = 0; i < stack.size(); i++)
		{
			result += stack.get(i);
		}
		System.out.println(result);
	} // main end
} // class end

### 괄호 (9012번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			String ps = st.nextToken();
			for (int j = 0; j < ps.length(); j++)
			{
				if (ps.charAt(j) == '(')
				{
					stack.push(j);
				}
				else if (ps.charAt(j) == ')')
				{
					if (stack.empty())
					{
						System.out.println("NO");
						break;
					}
					else
					{
						stack.pop();
					}
				}
				if (j == ps.length() - 1)
				{
					if (!(stack.empty()))
						System.out.println("NO");
					else 
						System.out.println("YES");
				}
			}
			stack.clear();
		}
	} // main end
} // class end
```

### 균형잡힌 세상 (4949번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	
		while (true)
		{
			Stack<String> stack = new Stack<String>();

			String temp = br.readLine();
			if (temp.charAt(0) == '.')
			{
				break;				
			}
			for (int i = 0; i < temp.length(); i++)
			{
				if (temp.charAt(i) == '(')
				{
					stack.push("(");
				}
				else if (temp.charAt(i) == ')')
				{
					if (stack.empty() || !(stack.lastElement().equals("(")))
						{
							System.out.println("no");
							break;
						}
						else
						{
							stack.pop();
						}
				}
				if (temp.charAt(i) == '[')
				{
					stack.push("[");
				}
				else if (temp.charAt(i) == ']')
				{
					if (stack.empty() || !(stack.lastElement().equals("[")))
						{
							System.out.println("no");
							break;
						}
						else
						{
							stack.pop();
						}
				}
				if (i == temp.length() - 1)
					{
						if (!(stack.empty()))
							System.out.println("no");
						else 
							System.out.println("yes");
					}
			}
			stack.clear();
		} // while 종료
	} // main end
} // class end
```

### 스택 수열 (1874번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int N = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		StringBuilder sb = new StringBuilder();
		st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int input = 0;
		for (int i = 1; i <= n; i++)
		{
			stack.add(i);
			input = i;
			sb.append("+" + "\n");
		}
		stack.pop();
		sb.append("-" + "\n");
		for (int i = 1; i < N; i++)
		{
			st = new StringTokenizer(br.readLine());
			n = Integer.parseInt(st.nextToken());
			int temp = input;
			for (int j = input + 1; j <= n; j++)
			{
				sb.append("+" + "\n");
				stack.add(j);
				temp = j;
			}
			input = temp;
			if (stack.lastElement() == n)
			{
				sb.append("-" + "\n");
				stack.pop();
			}
			else
			{
				System.out.println("NO");
				return;
			}
		}
		System.out.println(sb);
	} // main end
} // class end
```

### 오큰수 (17298번)

> 시간초과 코드 전부 스택에 넣은 뒤 스택 2개로 오큰수 구하기.

```java

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Stack;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int N = Integer.parseInt(st.nextToken());
		Stack<Integer> stack = new Stack<Integer>();
		Stack<Integer> stack2 = new Stack<Integer>();
		StringBuilder sb = new StringBuilder();
		st = new StringTokenizer(br.readLine());
		for (int i = 0; i < N; i++)
		{
			stack.add(Integer.parseInt(st.nextToken()));
		}
		int max = stack.peek();
		sb.append("-1");
		stack2.add(stack.peek());
		stack.pop();
		for (int i = 1; i < N; i++)
		{
			StringBuilder sb2 = new StringBuilder();
			if (stack.peek() < stack2.peek())
			{
				max = stack2.peek();
				stack2.add(stack.peek());
				sb2.append(max + " " + sb);
				sb = sb2;
				stack.pop();
			}
			else if (stack.peek() >= stack2.peek())
			{
				if (stack2.size() > 1)
					stack2.pop();
				if (stack.peek() < stack2.peek())
				{
					max = stack2.peek();
					stack2.add(stack.peek());
					sb2.append(max + " " + sb);
					sb = sb2;
				}
				else
				{
					max = stack.peek();
					sb2.append("-1" + " " + sb);
					sb = sb2;
				}
				stack2.add(stack.peek());
				stack.pop();
			}
		}
		System.out.println(sb);
	} // main end
} // class end
```

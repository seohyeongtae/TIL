### 색종이 만들기 (2630번)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;


public class Main {
	
	static int blue = 0;
	static int white = 0;
	public static void main(String[] args) throws IOException {
		
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int N = Integer.parseInt(st.nextToken());
		int arr[][];
		arr = new int[N][N];
		for (int i = 0; i < N; i++)
		{
			st = new StringTokenizer(br.readLine());
			for (int j = 0; j < N; j++)
			{
				int a = Integer.parseInt(st.nextToken());
				arr[i][j] = a;
			}
		}
		find_paper(arr, N);
		System.out.println(white);
		System.out.println(blue);
	} // main end

	private static void find_paper(int[][] arr, int n) {
		int temp[][] = new int[n / 2][n / 2];
		int result = arr[0][0];
		int k = 0;
		int m = 0;
		if (check_paper(arr, n))
		{
			if (result == 1)
				blue++;
			else
				white++;
			return;
		}
		else
		{
			k = 0;
			for (int i = 0; i < n / 2; i++)
			{ 
				m = 0;
				for (int j = 0; j < n / 2; j++)
				{
					temp[k][m] = arr[i][j];
					m++;
				}
				k++;
			}
			find_paper(temp, n / 2);
			k = 0;
			for (int i = n / 2; i < n; i ++)
			{
				m = 0;
				for (int j = 0; j < n / 2; j ++)
				{
					temp[k][m] = arr[i][j];
					m++;
				}
				k++;
			}
			find_paper(temp, n / 2);
			k = 0;
			for (int i = 0; i < n / 2; i ++)
			{
				m = 0;
				for (int j = n / 2; j < n; j ++)
				{
					temp[k][m] = arr[i][j];
					m++;
				}
				k++;
			}
			find_paper(temp, n / 2);
			k = 0;
			for (int i = n / 2; i < n; i ++)
			{
				m = 0;
				for (int j = n / 2; j < n; j ++)
				{
					temp[k][m] = arr[i][j];
					m++;
				}
				k++;
			}
			find_paper(temp, n / 2);
		}

	} // find_paper end

	private static boolean check_paper(int[][] arr, int n) {

		int result = arr[0][0];
		for (int i = 0; i < n; i ++)
		{
			for (int j = 0; j < n; j ++)
			{
				if (result != arr[i][j])
					return false;
			}
		}
		return true;
	}
	
} // class end
```

```java
import java.util.*;
import java.io.*;

public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static int[][] map;
    static int White = 0, Blue = 0;
    public static void main(String[] args) throws IOException{
     
        int N = Integer.parseInt(br.readLine());
        map = new int[N][N];

        for(int i = 0; i < N; i++){
            StringTokenizer st = new StringTokenizer(br.readLine());
            for(int j = 0; j < N; j++){
                map[i][j] = Integer.parseInt(st.nextToken());
            }
        }
        DAQ(map, 0, 0, N);

        System.out.println(White);
        System.out.println(Blue);
    }

    static void DAQ(int[][] arr, int y, int x, int S){
        
        int white = 0;
        int blue = 0;

        for(int i = y; i < y + S; i++){
            for(int j = x; j < x + S; j++){
                if(arr[i][j] == 1){
                    blue++;
                }else{
                    white++;
                }
            }
        }

        if(blue == 0){
            White++;
            return;
        }else if(white == 0){
            Blue++;
            return;
        }

        int nS = S / 2;

        DAQ(arr, y, x, nS);
        DAQ(arr, y, x + nS, nS);
        DAQ(arr, y + nS, x, nS);
        DAQ(arr, y + nS, x + nS, nS);
    }
}
```

### 피보나치 함수  (1003번)

> 재귀를 이용한 피보나치 수열은 시간 복잡도가 너무 안좋기 때문에
>
> 따로 캐쉬를 이용해서 계산하면 빨라진다 (재귀를 통한 중복 계산을 하지 않기 위해)
>
> ex ) fibo(7) 의 경우 fibo (1)을 13번 실행한다.

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;
 
public class Main {
 
	static int cache[];
	public static void main(String[] args) throws IOException {
 
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int num = Integer.parseInt(st.nextToken());
		int n;
		n = 0;
		cache = new int [41];
		cache[0] = 0;
		cache[1] = 1;
		for (int i = 0; i < num; i++)
		{
			st = new StringTokenizer(br.readLine());
			n = Integer.parseInt(st.nextToken());
			fibo(n);
			if (n == 0)
			{
				System.out.println("1 0");
			} 
			else
				System.out.println(cache[n - 1] + " " + cache[n]);
		}
	}

	private static void fibo(int n) 
	{
		for (int i = 2; i <= n; i++)
			cache[i] = cache[i - 1] + cache[i - 2];
	}
}
```



### 신나는 함수 실행 (9184번)

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.HashMap;
import java.util.StringTokenizer;
 
public class Main {
 
	static HashMap<String, Integer> w = new HashMap<String, Integer>();
	
	public static void main(String[] args) throws IOException {
 
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = 0;
		int b = 0;
		int c = 0;
		int result;
		a = Integer.parseInt(st.nextToken());
		b = Integer.parseInt(st.nextToken());
		c = Integer.parseInt(st.nextToken());
		while (!(a == -1 && b == -1 && c == -1))
		{
			result = cal(a, b, c);
			System.out.println("w("+a+", "+b+", "+c+") = "+result);
			st = new StringTokenizer(br.readLine());
			a = Integer.parseInt(st.nextToken());
			b = Integer.parseInt(st.nextToken());
			c = Integer.parseInt(st.nextToken());
		}
	}
    
	private static int cal(int a, int b, int c) 
	{
		String key = "w("+a+", "+b+", "+c+") = ";
		
		if (!(w.containsKey(key)))
		{
			if (a <= 0 || b <= 0 || c <= 0)
				w.put(key, 1);
			else if (a > 20 || b > 20 || c > 20)
				w.put(key, (cal(20, 20, 20)));
			else if (a < b && b < c)
				w.put(key, (cal(a, b, c-1) + cal(a, b-1, c-1) - cal(a, b-1, c)));
			else 
				w.put(key, (cal(a-1, b, c) + cal(a-1, b-1, c) + cal(a-1, b, c-1) - cal(a-1, b-1, c-1)));
		}
		return w.get(key);
	}
}
```

			 return 1;

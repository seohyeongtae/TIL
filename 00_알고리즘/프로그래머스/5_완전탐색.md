### 모의고사 (1번) 

```java
import java.util.*;

class Solution {
    public int[] solution(int[] answers) {

        // 1 -> 1, 2, 3, 4, 5       5
        // 2 -> 2, 1, 2, 3, 2, 4, 2, 5       8
        // 3 -> 3, 3, 1, 1, 2, 2, 4, 4, 5, 5        10
        int max = 0;
        int[] one = {1,2,3,4,5};
        int[] two = {2,1,2,3,2,4,2,5};
        int[] three = {3,3,1,1,2,2,4,4,5,5};

        int cnt1 = 0;
        int cnt2 = 0;
        int cnt3 = 0;

        int i = 0;
        int j1 = 0;
        int j2 = 0;
        int j3 = 0;
        while (i < answers.length)
        {
            if (answers[i] == one[j1])
                cnt1++;
            if (answers[i] == two[j2])
                cnt2++;
            if (answers[i] == three[j3])
                cnt3++;
            if (j1 == 4)
                j1 = -1;
            if (j2 == 7)
                j2 = -1;
            if (j3 == 9)
                j3 = -1;
            j1++;
            j2++;
            j3++;
            i++;
        }
        max = Math.max(cnt1, Math.max(cnt2, cnt3));
        ArrayList result = new ArrayList<Integer>();
        if (max == cnt1)
            result.add(1);
        if (max == cnt2)
            result.add(2);
        if (max == cnt3)
            result.add(3);
        int answer[] = new int[result.size()];
        for (int n = 0; n < result.size(); n++)
        {
            answer[n] = (int)result.get(n);
        }
        return answer;
    }
}
```

```java
import java.util.*;

class Solution {
    public static int[] solution(int[] answers) {
        int[][] patterns = {
                {1, 2, 3, 4, 5},
                {2, 1, 2, 3, 2, 4, 2, 5},
                {3, 3, 1, 1, 2, 2, 4, 4, 5, 5}
        };

        int[] hit = new int[3];
        for(int i = 0; i < hit.length; i++) {
            for(int j = 0; j < answers.length; j++) {
                if(patterns[i][j % patterns[i].length] == answers[j]) hit[i]++;
            }
        }

        int max = Math.max(hit[0], Math.max(hit[1], hit[2]));
        List<Integer> list = new ArrayList<>();
        for(int i = 0; i < hit.length; i++)
            if(max == hit[i]) list.add(i + 1);

        int[] answer = new int[list.size()];
        int cnt = 0;
        for(int num : list)
            answer[cnt++] = num;
        return answer;
    }
}
```

### 소수찾기 (2번)

```java
import java.util.*;

class Solution {
    int answer = 0;
    int t_len;
    boolean visit[];
    HashSet hs = new HashSet<Integer>();
    public int solution(String numbers) {
        t_len = numbers.length();
        String arr[] = new String[t_len];
        
        
        for (int i = 0; i < t_len; i++)
            arr[i] = numbers.substring(i, i + 1);
        visit = new boolean[t_len];
        make_num(arr, "", 0);
        return answer;
    } // solution end
    
    private void make_num(String[] arr, String result, int start)
    {
        if (start == t_len)
            return;
        
        for (int i = 0; i < t_len; i++)
        {
            if (visit[i] == false)
            {
                result += arr[i];
                visit[i] = true;
              //  System.out.println(result);
                check_num(result);
                make_num(arr, result, start + 1);
                result = result.substring(0, start);
                visit[i] = false;
            }
            
        }
        return;
    } // make_num end
    
    private void check_num(String result)
    {
        int temp = 0;
        for (int i = 0; i < result.length(); i++)
        {
            temp *= 10;
            temp += result.charAt(i) - '0';
        }
        if (temp != 0 && temp != 1)
        {
            if(!hs.contains(temp))
            {
                hs.add(temp); 
                // System.out.println(temp);
                for (int i = 3; i <= Math.sqrt(temp); i++)
                {
                    if (temp % i == 0)
                        return;

                }
                answer++;
            }
           
        }
        return;
    }
}
```

```java
import java.util.HashSet;
class Solution {
    public int solution(String numbers) {
        HashSet<Integer> set = new HashSet<>();
        permutation("", numbers, set);
        int count = 0;
        while(set.iterator().hasNext()){
            int a = set.iterator().next();
            set.remove(a);
            if(a==2) count++;
            if(a%2!=0 && isPrime(a)){
                count++;
            }
        }        
        return count;
    }

    public boolean isPrime(int n){
        if(n==0 || n==1) return false;
        for(int i=3; i<=(int)Math.sqrt(n); i+=2){
            if(n%i==0) return false;
        }
        return true;
    }

        public void permutation(String prefix, String str, HashSet<Integer> set) {
        int n = str.length();
        //if (n == 0) System.out.println(prefix);
        if(!prefix.equals("")) set.add(Integer.valueOf(prefix));
        for (int i = 0; i < n; i++)
          permutation(prefix + str.charAt(i), str.substring(0, i) + str.substring(i+1, n), set);

    }
}
```

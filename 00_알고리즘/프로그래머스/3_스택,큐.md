```java
import java.util.*;

class Solution {
    public int[] solution(int[] progresses, int[] speeds) {
        int[] answer = {};
        
        Queue program = new LinkedList<Integer>();
        Queue speed = new LinkedList<Integer>();
        for (int i = 0; i < progresses.length; i++)
        {
           program.add(progresses[i]);
            speed.add(speeds[i]);
        }
        int days = 0;
        int num = 0;
        int size = 0;
        while (program.peek() != null)
        {
            while ((int)program.peek() + ((int)speed.peek() * days) < 100)
            {
                days++;
                num = 0;
            }
           
            while (program.peek() != null && 
                   (int)program.peek() + ((int)speed.peek() * days) >= 100)
            {
                program.poll();
                speed.poll();
                num++;
            }
            size++;
           answer = make_answer(answer, size , num);
            
        }
        return answer;
    } // solution end
    
    private int[] make_answer(int[] answer, int size, int num)
    {
        int[] temp = new int[size];
        for (int i = 0; i < answer.length; i++)
        {
            temp[i] = answer[i];
        }
        temp[size - 1] = num;
        return temp;
    }
}
```

```java
import java.util.*;

class Solution {
    static int result;
    public int solution(int[] priorities, int location) {
        int answer = 0;
        int[] temp = new int[priorities.length];
        int output = 0;
        result = location;
        int len = priorities.length;
        Queue arr = new LinkedList<Integer>();
        
        if (len == 1)
            return 1;
        for (int i = 0; i < len; i++)
        {
            temp[i] = priorities[i];
            arr.add(priorities[i]);
        }
        while (arr.peek() != null)
        {
            int check = 0;
            for (int i = 0; i < temp.length; i++)
            {
                if ((int)arr.peek() < temp[i])
                {
                    temp = change_temp(temp, result);
                    arr.add(arr.poll());
                    check = 1;
                    break;
                }
            }
           if (check == 0)
           {
               output++;
               arr.poll();
               if (result == 0)
                   return output;
               temp = delete_temp(temp);
           }
        } // while end
        
        return output;
    } // solution end
    
    private int[] delete_temp(int[] temp)
    {
        int[] a = new int[temp.length - 1];
        result -= 1;
        for (int i = 0; i < temp.length - 1; i++)
        {
            a[i] = temp[i + 1];
        }
        return a;
    }
    
    private int[] change_temp(int[] temp, int t)
    {
        int[] a = new int[temp.length];
        if (t == 0)
        {
            result = temp.length - 1;
        }
        else
            result = result - 1;
        
        for (int i = 0; i < temp.length - 1; i++)
        {
            a[i] = temp[i + 1];
        }
        a[temp.length - 1] = temp[0];
        return a;
    }
}
```

```java
import java.util.*;

class Solution {
    public int solution(int[] priorities, int location) {
        int answer = 0;
        int l = location;

        Queue<Integer> que = new LinkedList<Integer>();
        for(int i : priorities){
            que.add(i);
        }

        Arrays.sort(priorities);
        int size = priorities.length-1;



        while(!que.isEmpty()){
            Integer i = que.poll();
            if(i == priorities[size - answer]){
                answer++;
                l--;
                if(l <0)
                    break;
            }else{
                que.add(i);
                l--;
                if(l<0)
                    l=que.size()-1;
            }
        }

        return answer;
    }
}
```

### 다리를 지나는 트럭 (3번)

```java
import java.util.*;

class Solution {
    public int solution(int bridge_length, int weight, int[] truck_weights) {
        int answer = 0;
        Queue truck = new LinkedList<Integer>();
        Queue truck_time = new LinkedList<Integer>();
        
        truck_time.add(bridge_length);
        truck.add(0);
       
        int len = truck_weights.length;
        int t_weights = truck_weights[0];
        
        int time = 0;
        int i = 1;
        if (len == 1)
            return bridge_length + 1;
        while (len > 1)
        {
            time++;
            if ((int)truck_time.peek() == time)
            {
                t_weights -= truck_weights[(int)truck.poll()];
                truck_time.poll();
            }
            if (i < len && weight >= t_weights + truck_weights[i])
            {
                
                t_weights += truck_weights[i];
                truck_time.add(time + bridge_length);
                truck.add(i);
                i++;
            }
            if (truck.peek() == null)
                break;
       
        } // while end
        return time + 1;
    }
}
```

```java
import java.util.*;

class Solution {
    class Truck {
        int weight;
        int move;

        public Truck(int weight) {
            this.weight = weight;
            this.move = 1;
        }

        public void moving() {
            move++;
        }
    }

    public int solution(int bridgeLength, int weight, int[] truckWeights) {
        Queue<Truck> waitQ = new LinkedList<>();
        Queue<Truck> moveQ = new LinkedList<>();

        for (int t : truckWeights) {
            waitQ.offer(new Truck(t));
        }

        int answer = 0;
        int curWeight = 0;

        while (!waitQ.isEmpty() || !moveQ.isEmpty()) {
            answer++;

            if (moveQ.isEmpty()) {
                Truck t = waitQ.poll();
                curWeight += t.weight;
                moveQ.offer(t);
                continue;
            }

            for (Truck t : moveQ) {
                t.moving();
            }

            if (moveQ.peek().move > bridgeLength) {
                Truck t = moveQ.poll();
                curWeight -= t.weight;
            }

            if (!waitQ.isEmpty() && curWeight + waitQ.peek().weight <= weight) {
                Truck t = waitQ.poll();
                curWeight += t.weight;
                moveQ.offer(t);
            }
        }

        return answer;
    }
}
```

### 디스크 컨트롤러 (2번)

> 람다식 사용

```java
import java.util.*;

class Solution {
    public int solution(int[][] jobs) {
        int answer = 0;
        
        int len = jobs.length;
 
        // Arrays.sort(jobs, new Comparator<int[]>(){
        //    @Override 
        //     public int compare (int[] o1, int[] o2){
        //         if (o1[0] == o2[0])
        //         {
        //             return o1[1] - o2[1];
        //         }
        //         else
        //         {
        //             return o1[0] - o2[0];
        //         }
        //     }
        // });
        Arrays.sort(jobs, (o1, o2) -> o1[0] - o2[0]);
        
		int time = 0;
		int idx = 0;
        // heap 이용할 때 우선순위로 정렬한뒤 jobs[0] (요청시간)을 뺀다.
        // Queue or PriorityQueue 의 비교대상을 지정해주어야 PriorityQueue를 사용할 수 있다.
		PriorityQueue<int[]> q = new PriorityQueue<>((o1, o2) -> o1[1] - o2[1]);

		while (idx < len || !q.isEmpty()) {
			while (idx < len && jobs[idx][0] <= time)
				q.offer(jobs[idx++]);
			// 시작할 수 있는 작업이 없다면 해야 할 작업 중 가장 먼저 해야 할 작업의 요청 시간으로 현재 시간을 바꾸고, 큐에 삽입. 
            // 뒤에 time - job[0] 을 해주기 때문에 요청시간이 매우 큰 경우도 해결가능. 
			if (q.isEmpty())
				time = jobs[idx][0];
			else {
				int[] job = q.poll();
				answer += time - job[0] + job[1];
				time += job[1];
			}
		}
        return answer / len;
    }
}
```

### 이중우선순위큐 (3번)

```java
import java.util.*;

class Solution {
    public int[] solution(String[] operations) {
        int[] answer = new int[2];
        answer[0] = 0;
        answer[1] = 0;
        PriorityQueue max = new PriorityQueue<Integer>((o1, o2) -> o2 - o1);
        PriorityQueue min = new PriorityQueue<Integer>();
        int max_count = 0;
        int min_count = 0;
        int count = 0;
        
        for (int i = 0; i < operations.length; i++)
        {
            String temp[] = operations[i].split(" ");
            if (temp[0].equals("I"))
            {
                max.offer(Integer.parseInt(temp[1]));
                min.offer(Integer.parseInt(temp[1]));
                count++;
            }
            else if (temp[0].equals("D"))
            {
                 //System.out.println(temp[1].substring(0,1));
                if (temp[1].equals("1"))
                {
                    if (max.peek() != null )
                    {
                        max.remove();
                        max_count++;
                    }
                    else
                        continue;
                    
                }
                else if (temp[1].substring(0,1).equals("-"))
                {
                    if (min.peek() != null)
                    {
                        min.remove();
                        min_count++;
                    }
                    else
                        continue;
                }
            }
            if (count - max_count - min_count == 0)
            {
                while (min.peek() != null)
                {
                    min.remove();
                }
                while (max.peek() != null)
                {
                    max.remove();
                }
            }
        }
      //  System.out.println(count - max_count - min_count);
        if (count - max_count - min_count <= 1)
        {
            return answer;
        }
        answer[0] = (int)max.peek();
        answer[1] = (int)min.peek();
        return answer;
    }
}
```

```java
import java.util.Collections;
import java.util.PriorityQueue;

class Solution {
    public int[] solution(String[] arguments) {
        int[] answer = {0,0};

        PriorityQueue<Integer> pq = new PriorityQueue<Integer>();
        PriorityQueue<Integer> reverse_pq = new PriorityQueue<Integer>(Collections.reverseOrder());

        for(int i=0; i<arguments.length; i++) {
            String temp[] = arguments[i].split(" ");
            switch(temp[0]) {
            case "I" : 
                pq.add(Integer.parseInt(temp[1]));
                reverse_pq.add(Integer.parseInt(temp[1]));
                break;
            case "D" :
                if(pq.size() > 0) {
                    if(Integer.parseInt(temp[1]) == 1) {
                        // 최댓값 삭제
                        int max = reverse_pq.poll();
                        pq.remove(max);
                    } else {
                        // 최솟값 삭제
                        int min = pq.poll();
                        reverse_pq.remove(min);
                    }
                }
                break;
            }
        }

        if(pq.size() >= 2) {
            answer[0] = reverse_pq.poll();
            answer[1] = pq.poll();
        }

        System.out.println(answer[0] + ":" + answer[1]);

        return answer;
    }
}
```

class Solution {
    public int solution(int[] scoville, int K) {
        int answer = 0;
        PriorityQueue sc = new PriorityQueue<Integer>();
        for (int i = 0; i < scoville.length; i++)
        {
            sc.offer(scoville[i]);
        }
        while ((int)sc.peek() < K)
        {
            int first = (int)sc.poll();
            if (sc.peek() == null)
                return -1;
            int second = (int)sc.poll();
            sc.offer(first + (second * 2));

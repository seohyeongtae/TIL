### N으로 표현 (1번) 

```java
import java.util.Set;
import java.util.HashSet;
import java.util.List;
import java.util.ArrayList;

class Solution {
    public int solution(int N, int number) {
    int answer = -1;
        List<Set<Long>> arr = new ArrayList<>();

        arr.add(null);
        arr.add(new HashSet<>());

        arr.get(1).add(new Long(N));

        for(int i=1; i<=8; i++){
            if(i>=2){
                arr.add(new HashSet<>());

                StringBuilder num = new StringBuilder();
                for(int j=0;j<i;j++) {
                    num.append(N);
                }
                arr.get(i).add(Long.parseLong(num.toString()));

                for(int j=1; j<i; j++ ){               
                    for(long k : arr.get(j)){
                        for(long l : arr.get(i-j)){
                            arr.get(i).add(k+l);
                            arr.get(i).add(k-l);
                            arr.get(i).add(k*l);
                            if(l!=0){
                                arr.get(i).add(k/l);
                            }
                        }
                    }
                }                
            }

            if(arr.get(i).contains((long)number)){
                return i;
            }
        }
        return answer;
    }
}
```

### 정수 삼각형

```java
import java.util.*;

class Solution {
    public int solution(int[][] triangle) {
        int answer = 0;
        // 같은 수나 +1 만 더할 수 있다.
        // 가장 큰 수만 남긴다.
        // 깊이는 triangle 의 길이
        triangle = dfs(triangle, triangle.length - 1);
        answer = triangle[0][0];
        return answer;
    }
    
    private int[][] dfs (int[][] triangle, int depth)
    {
        if (depth >= 0)
        {
            for(int i = 0; i < depth; i++)
                triangle[depth - 1][i] = Math.max(triangle[depth - 1][i] + triangle[depth][i], triangle[depth - 1][i] + triangle[depth][i + 1]);
 
            dfs(triangle, depth - 1);
        }
        
        return triangle;
    }
    
} // solution end
```

import java.util.*;
class Solution {
    public int solution(int[][] triangle) {
        for (int i = 1; i < triangle.length; i++) {
            triangle[i][0] += triangle[i-1][0];
            triangle[i][i] += triangle[i-1][i-1];
            for (int j = 1; j < i; j++) 
                triangle[i][j] += Math.max(triangle[i-1][j-1], triangle[i-1][j]);
        }

        return Arrays.stream(triangle[triangle.length-1]).max().getAsInt();
    }
}

### 등굣길 (3번)

```java
class Solution {
    public int solution(int m, int n, int[][] puddles) {
        int answer = 0;
        int di = 1000000007;
        int [][] arr = new int[m][n];
        arr[0][0] = 1;
        for (int i = 0; i < m; i++)
        {
            if (!check_puddles(i,0,puddles))
                break;    
            arr[i][0] = 1;   
        }
        for (int i = 0; i < n; i++)
        {
            if (!check_puddles(0,i,puddles))
                break;
            arr[0][i] = 1; 
        }
        
        for (int i = 1; i < m; i++)
        {
            for (int j = 1; j < n; j++)
            {
               if (check_puddles(i,j,puddles))
                    arr[i][j] = (arr[i-1][j]  + arr[i][j-1]) % di;
              //  System.out.println(arr[i][j]);
            }
        }
        answer = arr[m-1][n-1];
        return answer;
    }
    
    private boolean check_puddles(int i, int j, int[][] puddles)
    {
        for (int k = 0; k < puddles.length; k++)
        {
            int x = puddles[k][0];
            int y = puddles[k][1];
            if ((i == x - 1  && j == y - 1))
                return false;
        }
        return true;
    }
} // solution end
```

class Solution {
    public int solution(int m, int n, int[][] puddles) {
        int[][] dp = new int[m+1][n+1];
        for(int i=0;i<puddles.length;i++){
            dp[puddles[i][0]][puddles[i][1]]=-1;
        }
        dp[1][1]=1; 
        for(int i=1;i<=m;i++){
            for(int j=1;j<=n;j++){
                if(dp[i][j]==-1){
                    dp[i][j]=0;
                    continue;
                }
                if(i!=1)    dp[i][j]+=dp[i-1][j]%1000000007;
                if(j!=1)    dp[i][j]+=dp[i][j-1]%1000000007;
            }
        }
        return dp[m][n]%1000000007;
    }
}

### 도둑질 (4번)

```java
class Solution {
    public int solution(int[] money) {
        int answer = 0;
        int len = money.length;
        int[] dp1 = new int[len];
        int[] dp2 = new int[len];
        
        for (int i = 0; i < len - 1; i++)
        {
            if (i == 0)
                dp1[i] = money[i];
            else if (i == 1)
                dp1[i] = money[i - 1];
            else
                dp1[i] = Math.max(dp1[i - 2] + money[i], dp1[i - 1]);
            answer = Math.max(answer, dp1[i]);
        }
        for (int i = 0; i < len; i++)
        {
            if (i == 0)
                dp2[i] = 0;
            else if (i == 1)
                dp2[i] = money[i];
            else
                dp2[i] = Math.max(dp2[i - 2] + money[i], dp2[i - 1]);
            answer = Math.max(answer, dp2[i]);
        }
        return answer;
    }// solution end
}
```

class Solution {
    public int solution(int[] money) {        
        int[][] pick = new int[2][money.length];

        pick[0][0] = money[0];
        pick[0][1] = money[0];
        pick[1][0] = 0;
        pick[1][1] = money[1];

        for(int i=2; i<money.length; i++) {
            pick[0][i] = Math.max(pick[0][i-2] + money[i], pick[0][i-1]);
            pick[1][i] = Math.max(pick[1][i-2] + money[i], pick[1][i-1]);
        }

        return Math.max(pick[0][pick[0].length-2], pick[1][pick[1].length-1]);
    }
}

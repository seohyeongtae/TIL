## 동적계획법

### N으로 표현 (1번) 

```java
import java.util.Set;
import java.util.HashSet;
import java.util.List;
import java.util.ArrayList;

class Solution {
    public int solution(int N, int number) {
    int answer = -1;
        List<Set<Long>> arr = new ArrayList<>();

        arr.add(null);
        arr.add(new HashSet<>());

        arr.get(1).add(new Long(N));

        for(int i=1; i<=8; i++){
            if(i>=2){
                arr.add(new HashSet<>());

                StringBuilder num = new StringBuilder();
                for(int j=0;j<i;j++) {
                    num.append(N);
                }
                arr.get(i).add(Long.parseLong(num.toString()));

                for(int j=1; j<i; j++ ){               
                    for(long k : arr.get(j)){
                        for(long l : arr.get(i-j)){
                            arr.get(i).add(k+l);
                            arr.get(i).add(k-l);
                            arr.get(i).add(k*l);
                            if(l!=0){
                                arr.get(i).add(k/l);
                            }
                        }
                    }
                }                
            }

            if(arr.get(i).contains((long)number)){
                return i;
            }
        }
        return answer;
    }
}
```

### 정수 삼각형

```java
import java.util.*;

class Solution {
    public int solution(int[][] triangle) {
        int answer = 0;
        // 같은 수나 +1 만 더할 수 있다.
        // 가장 큰 수만 남긴다.
        // 깊이는 triangle 의 길이
        triangle = dfs(triangle, triangle.length - 1);
        answer = triangle[0][0];
        return answer;
    }
    
    private int[][] dfs (int[][] triangle, int depth)
    {
        if (depth >= 0)
        {
            for(int i = 0; i < depth; i++)
                triangle[depth - 1][i] = Math.max(triangle[depth - 1][i] + triangle[depth][i], triangle[depth - 1][i] + triangle[depth][i + 1]);
 
            dfs(triangle, depth - 1);
        }
        
        return triangle;
    }
    
} // solution end
```

import java.util.*;
class Solution {
    public int solution(int[][] triangle) {
        for (int i = 1; i < triangle.length; i++) {
            triangle[i][0] += triangle[i-1][0];
            triangle[i][i] += triangle[i-1][i-1];
            for (int j = 1; j < i; j++) 
                triangle[i][j] += Math.max(triangle[i-1][j-1], triangle[i-1][j]);
        }

        return Arrays.stream(triangle[triangle.length-1]).max().getAsInt();
    }
}

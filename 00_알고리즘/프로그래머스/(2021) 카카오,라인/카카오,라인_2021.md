package baekjoon;

import java.util.HashMap;
import java.util.StringTokenizer;

class Solution {
    public int[] solution(String[] id_list, String[] report, int k) {
        int[] answer = {};
        int id_length = id_list.length;
        HashMap hm = new HashMap<String, Integer>();
        HashMap arr_report = new HashMap<String, Integer>();
        int[][] arr = new int[id_list.length][id_list.length];
        for (int i = 0; i < id_length; i++)
        {
        	hm.put(id_list[i], i);
        }
        for (int i = 0; i < report.length; i++)
        {
        	StringTokenizer st = new StringTokenizer(report[i]);
        	String first = st.nextToken();
        	String second = st.nextToken();
        	
        	if (arr[(int) hm.get(first)][(int) hm.get(second)] == 0)
        	{
        		arr[(int) hm.get(first)][(int) hm.get(second)] = 1;	
        	}
        }
        for (int i = 0; i < id_length; i++)
        {
        	for (int j = 0; j < id_length; j++)
        	{
        		if (arr[i][j] == 1)
        		{
        			arr_report.put(id_list[j],  (int) arr_report.getOrDefault(id_list[j], 0) + 1);
        		}
        	}
        }
        
        boolean[] result = new boolean[id_length];
        for (int i = 0; i < id_length; i++)
        {
        	if (arr_report.containsKey(id_list[i]))
        	{
        		if ((int)arr_report.get(id_list[i]) >= k)
        		{
        			result[i] = true;
        		}
        	}
        }
        
        int[] final_result = new int[id_length];
        for (int i = 0; i < id_length; i++)
        {
        	for (int j = 0; j < id_length; j++)
        	{
        		if (arr[i][j] == 1 && result[j] == true)
        		{
        			final_result[i] += 1;
        		}
        	}
        }
        answer = final_result;
        return answer;
    }
}

    public int[] solution(int[] fees, String[] records) {

        		hm.put(car_num, car_time);
        	}
        	else
        	{
        		String t = hm.get(car_num);
        		String[] first = t.split(":");
        		String[] second = car_time.split(":");
        		int precent_time = make_time(first, second);
        		hm_time.put(car_num, hm_time.getOrDefault(car_num, 0) + precent_time);
        		hm.remove(car_num);
        	}
        }
        
        if (!(hm.isEmpty()))
        {
        	for (String c : hm.keySet())
        	{
        		String[] first = hm.get(c).split(":");
        		String[] second = {"23","59"};
        		int precent_time = make_time(first, second);
        		hm_time.put(c, hm_time.getOrDefault(c, 0) + precent_time);
        	}
        }
        hm.clear();
        for (String c : hm_time.keySet())
        	al.add(c);
        Collections.sort(al);
        int result[] = new int[al.size()];
        for (int i = 0; i < al.size(); i++)
        {
        	int total_fee = 0;
        	total_fee += fee;
        	if (hm_time.get(al.get(i)) < time)
        	{
        		result[i] = total_fee;
        	}
        	else
        	{
        		int t = (hm_time.get(al.get(i)) - time) / unit_time;
        		if ((hm_time.get(al.get(i)) - time) % unit_time != 0 )
        			t += 1;
        		total_fee += t * unit_fee;
        		result[i] = total_fee;
        	}
        }
        answer = result;
        return answer;
    }

	private int make_time(String[] first, String[] second) {
		// 0 : hour  1 : min
		// second - first
		int first_hour[] = new int[2];
		int second_hour[] = new int[2];
		
		for (int i = 0; i < 2; i++)
		{
			for (int j = 0; j < 2; j++)
			{
				first_hour[i] *= 10;
				first_hour[i] += first[i].charAt(j) +'0';
				second_hour[i] *= 10;
				second_hour[i] += second[i].charAt(j) +'0';
			}
		}
		int a = 0;
		int b = 0;
		a = (first_hour[0] * 60) + first_hour[1];
		b = (second_hour[0] * 60) + second_hour[1];
		
		return b - a;
	}
}

    public String solution(String[] research, int n, int k) {
        String answer = "";

        //2 * n * k    a = 8

        	}
        }
        String[] check = new String[hm.size()];
        int ss = 0;
        for (String c : hm.keySet())
        {
        	check[ss] = c;
        	ss++;
        }
        
        int[][] serch = new int [hm.size()][research.length];
        for (int i = 0; i < research.length; i++)
        {
        	for (int j = 0; j < research[i].length(); j++)
        	{
        		String c = research[i].substring(j, j + 1);
//        		System.out.println(c);
        		for (int a = 0; a < hm.size(); a++)
        		{

        			if (check[a].equals(c))
        			{
        				serch[a][i] += 1;
        			}
        		}
        	}
        }
        // n = 연속된 날
        // k = 날마다 k번 이상 검색
        // 같은 연속된 일동안  n * k * 2 만큼 재생
        int min = n * k * 2;
        int [] final_check = new int [serch.length];
        for (int i = 0; i < serch.length; i++)

        	for (int j = 0; j <= serch[i].length - n; j++)

        		{
        			middle += serch[i][m];
        			if (serch[i][m] < k)
        				break;
        		}
        		if (middle >= min)

        		final_result = check[i];

### 오랜 기간 보호한 동물(1)

```
/* 
  - NOT IN 사용 코드
SELECT NAME,DATETIME from ANIMAL_INS WHERE ANIMAL_ID NOT IN (
    SELECT ANIMAL_ID FROM ANIMAL_OUTS
)
ORDER BY DATETIME ASC LIMIT 3
*/

SELECT * FROM (
SELECT A1.NAME, A1.DATETIME
FROM ANIMAL_INS A1, ANIMAL_OUTS A2
WHERE A1.ANIMAL_ID = A2.ANIMAL_ID(+)
AND A2.ANIMAL_ID IS NULL
ORDER BY A1.DATETIME)
WHERE ROWNUM <= 3;
```

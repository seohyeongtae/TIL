```java
import java.util.*;
class Solution {
    public int solution(int n, int[] lost, int[] reserve) {

        n = n - lost.length;
        HashSet hs_lost = new HashSet<Integer>();
        HashSet hs_reserve = new HashSet<Integer>();
        Arrays.sort(reserve);
        for (int i = 0; i < lost.length; i++)
            hs_lost.add(lost[i]);
        for (int i = 0; i < reserve.length; i++)
        {
            if (hs_lost.contains(reserve[i]))
            {
                n += 1;
                hs_lost.remove(reserve[i]);
                hs_reserve.add(reserve[i]);
            }
        }
        for (int i = 0; i < reserve.length; i++)
        {
            if (!hs_reserve.contains(reserve[i]) && hs_lost.contains(reserve[i] - 1))
            {
                n += 1;
                hs_lost.remove(reserve[i] - 1);
            }
            else if (!hs_reserve.contains(reserve[i]) &&
                     hs_lost.contains(reserve[i] + 1))
            {
                n += 1;
                hs_lost.remove(reserve[i] + 1);
            }
        }
        int answer = n;
        return answer;
    }
}
```

```java
class Solution {
    public int solution(int n, int[] lost, int[] reserve) {
        int[] people = new int[n];
        int answer = n;

        for (int l : lost) 
            people[l-1]--;
        for (int r : reserve) 
            people[r-1]++;

        for (int i = 0; i < people.length; i++) {
            if(people[i] == -1) {
                if(i-1>=0 && people[i-1] == 1) {
                    people[i]++;
                    people[i-1]--;
                }else if(i+1< people.length && people[i+1] == 1) {
                    people[i]++;
                    people[i+1]--;
                }else 
                    answer--;
            }
        }
        return answer;
    }
}
```

```java
class Solution {
    public int solution(String name) {
        int answer = 0;
        int len = name.length();
        int name_arr[] = new int [len];
        int arr[] = new int[len];
        boolean check[] = new boolean[len];
        int count = 0;
        for (int i = 0; i < len; i++)
        {
            int temp = name.charAt(i) - '0' - 17;
            if (temp < 14)
                name_arr[i] = temp;
            else
                name_arr[i] = 26 - temp;
            answer += name_arr[i];
            if (name_arr[i] != 0 && i > 0)
            {
                check[i] = true;
                count++;
            }
        } // for end
  
        if (len == 1)
            return name_arr[0];
        if (check_arr(name_arr, arr))
            return 0;
        
        
        int location = 0;
        int count_result = 0;
        while (count > 0)
        {
            int t = location;
            int t_t = 0;
            int k = location;
            int k_k = 0;
            while (check[t] == false)
            {
                t++;
                t_t++;
                if (t == len)
                    t = 0;
            }
            while (check[k] == false)
            {
                k--;
                k_k++;
                if (k < 0)
                    k = len - 1;
            }
            if (k_k >= t_t)
            {
                check[t] = false;
                count_result += t_t;
                location = t;
            }
            else
            {
                check[k] = false;
                count_result += k_k; 
                location = k;
            }
            count--;
        }
       // System.out.println(count_result);
        
        return answer + count_result;
    } // solution end
    
    private boolean check_arr(int [] name_arr, int [] arr)
    {
        for (int i = 0; i < name_arr.length; i++)
        {
            if (name_arr[i] != arr[i])
                return false;
        }
        return true;
    } // check arr end
}
```

```java
class Solution {
    public String solution(String number, int k) {
        StringBuilder a = new StringBuilder(number);
        int i = 0;
        int j = 0;
        int idx;
        int len;

        for (i = 0; i < k; i++) {
            len = a.length();
            idx = len - 1;
            for (j = 0; j < len - 1; j++) {
                if (a.charAt(j) < a.charAt(j + 1)) {
                    idx = j;
                    break;
                }
            }
            a.deleteCharAt(idx);
        }

        return a.toString();
    }
}
```

```java
import java.util.Stack;
class Solution {
    public String solution(String number, int k) {
        char[] result = new char[number.length() - k];
        Stack<Character> stack = new Stack<>();

        for (int i=0; i<number.length(); i++) {
            char c = number.charAt(i);
            while (!stack.isEmpty() && stack.peek() < c && k-- > 0) {
                stack.pop();
            }
            stack.push(c);
        }
        for (int i=0; i<result.length; i++) {
            result[i] = stack.get(i);
        }
        return new String(result);
    }
}
```

### 구명보트 (4번)

```java
import java.util.*;
class Solution {
    public int solution(int[] people, int limit) {
        int answer = 0;
        Arrays.sort(people);
        boolean check[] = new boolean[people.length];
        int count = 0;
        int check_count = 0;
        int len = people.length;
        for (int i = len - 1; i > 0; i--)
        {
            if (people[0] + people[i] <= limit)
                break;
            check[i]  = true;
            count++;
            check_count++;
        }
        int temp = len - 1 - check_count;
        int a = 0;
        for (int i = 0; i < len - check_count; i++)
        {
            a = 0;
            if (check[i] == true)
                continue;
            if (i < len - 1 && people[i] + people[i + 1] <= limit)
            {
                for (int j = temp; j > i; j--)
                 {
                    if (check[j] == true)
                        continue;
                    if (people[i] + people[j] <= limit)
                    {
                        check[i] = true;
                        check[j] = true;
                        count++;
                        temp = j;
                        a = 1;
                        break;
                    }
                }
            }
            if (a == 0)
            {
                check[i] = true;
                count++;
            }
        }
        answer = count;
        return answer;
    }
}
```

### 섬 연결하기 (5번)

```java
import java.util.*;

class Solution {
    private int get_parent(int[] parent, int x)
    {
        if (parent[x] == x)
            return x;
        return parent[x] = get_parent(parent, parent[x]);
    }
    
    private void union_parent(int[] parent, int a, int b)
    {
        a = get_parent(parent, a);
        b = get_parent(parent, b);
        if (a < b)
            parent[b] = a;
        else 
            parent[a] = b;
    }
    
    private boolean find_parent(int[] parent, int a, int b)
    {
        a = get_parent(parent, a);
        b = get_parent(parent, b);
        if (a == b)
            return false;
        return true;
    }

    public int solution(int n, int[][] costs) {
        int answer = 0;
        // 크루스칼 알고리즘
        int[] parent = new int[n];
        for (int i = 0; i < n; i++)
            parent[i] = i;
        Arrays.sort(costs, new Comparator<int[]>(){
            @Override
            public int compare(int o1[], int o2[])
            {
                return Integer.compare(o1[2], o2[2]);
            }
        });
        for (int i = 0; i < costs.length; i++)
        {
            if (find_parent(parent, costs[i][0], costs[i][1]))
            {
                union_parent(parent, costs[i][0], costs[i][1]);
                answer += costs[i][2];
            }
        }
            
        
        return answer;
    }
} // solution end
```

```java
import java.util.Arrays;

class Solution
{
    public int solution(int n, int[][] costs)
    {
        int sum = 0;
        int[] island = new int[n];

        for(int i = 0; i < n; i++)
            island[i] = i;

        Arrays.sort(costs, (a, b) -> Integer.compare(a[2], b[2]));

        for(int i = 0; i < costs.length; i++)
        {
            if(find(island, costs[i][0]) != find(island, costs[i][1]))
            {
                unite(island, costs[i][0], costs[i][1]);
                sum += costs[i][2];
            }
        }

        return sum;
    }

    private int find(int[] island, int x)
    {
        if(island[x]== x)
            return x;
        return find(island, island[x]);
    }

    private void unite(int[] island, int x, int y)
    {
        int a = find(island, x);
        int b = find(island, y);
        island[a] = b;
    }
}
```

```java
import java.util.*;
class Solution {
    public int solution(int[][] routes) {
        int answer = 1;
        // -30000 ~ 30000
        Arrays.sort(routes, new Comparator<int []>(){
            @Override
            public int compare(int[] o1, int[] o2)
            {
                if (o1[1] == o2[1])
                    return o1[0] - o2[0];
                return o1[1] - o2[1];
            }
        });
        // for (int i = 0; i < routes.length; i++)
        //     System.out.println(routes[i][0]);
        ArrayList min = new ArrayList<Integer>();
        min.add(routes[0][1]);
        for (int i = 1; i < routes.length; i++)
        {
            int a = 0;
            for (int j = 0; j < min.size(); j++)
            {
                int temp = (int)min.get(j);
                if (routes[i][0] <= temp && routes[i][1] >= temp)
                {
                    a = 1;
                    continue;
                }
            }
            if (a == 0)
            {
                min.add(routes[i][1]);
                answer += 1;    
            }
        }

        return answer;
    }
}
```

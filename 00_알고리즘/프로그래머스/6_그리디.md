### 체육복 (1번) 

```java
import java.util.*;
class Solution {
    public int solution(int n, int[] lost, int[] reserve) {

        n = n - lost.length;
        HashSet hs_lost = new HashSet<Integer>();
        HashSet hs_reserve = new HashSet<Integer>();
        Arrays.sort(reserve);
        for (int i = 0; i < lost.length; i++)
            hs_lost.add(lost[i]);
        for (int i = 0; i < reserve.length; i++)
        {
            if (hs_lost.contains(reserve[i]))
            {
                n += 1;
                hs_lost.remove(reserve[i]);
                hs_reserve.add(reserve[i]);
            }
        }
        for (int i = 0; i < reserve.length; i++)
        {
            if (!hs_reserve.contains(reserve[i]) && hs_lost.contains(reserve[i] - 1))
            {
                n += 1;
                hs_lost.remove(reserve[i] - 1);
            }
            else if (!hs_reserve.contains(reserve[i]) &&
                     hs_lost.contains(reserve[i] + 1))
            {
                n += 1;
                hs_lost.remove(reserve[i] + 1);
            }
        }
        int answer = n;
        return answer;
    }
}
```

```java
class Solution {
    public int solution(int n, int[] lost, int[] reserve) {
        int[] people = new int[n];
        int answer = n;

        for (int l : lost) 
            people[l-1]--;
        for (int r : reserve) 
            people[r-1]++;

        for (int i = 0; i < people.length; i++) {
            if(people[i] == -1) {
                if(i-1>=0 && people[i-1] == 1) {
                    people[i]++;
                    people[i-1]--;
                }else if(i+1< people.length && people[i+1] == 1) {
                    people[i]++;
                    people[i+1]--;
                }else 
                    answer--;
            }
        }
        return answer;
    }
}
```

### 조이스틱 (2번)

```java
class Solution {
    public int solution(String name) {
        int answer = 0;
        int len = name.length();
        int name_arr[] = new int [len];
        int arr[] = new int[len];
        boolean check[] = new boolean[len];
        int count = 0;
        for (int i = 0; i < len; i++)
        {
            int temp = name.charAt(i) - '0' - 17;
            if (temp < 14)
                name_arr[i] = temp;
            else
                name_arr[i] = 26 - temp;
            answer += name_arr[i];
            if (name_arr[i] != 0 && i > 0)
            {
                check[i] = true;
                count++;
            }
        } // for end
  
        if (len == 1)
            return name_arr[0];
        if (check_arr(name_arr, arr))
            return 0;
        
        
        int location = 0;
        int count_result = 0;
        while (count > 0)
        {
            int t = location;
            int t_t = 0;
            int k = location;
            int k_k = 0;
            while (check[t] == false)
            {
                t++;
                t_t++;
                if (t == len)
                    t = 0;
            }
            while (check[k] == false)
            {
                k--;
                k_k++;
                if (k < 0)
                    k = len - 1;
            }
            if (k_k >= t_t)
            {
                check[t] = false;
                count_result += t_t;
                location = t;
            }
            else
            {
                check[k] = false;
                count_result += k_k; 
                location = k;
            }
            count--;
        }
       // System.out.println(count_result);
        
        return answer + count_result;
    } // solution end
    
    private boolean check_arr(int [] name_arr, int [] arr)
    {
        for (int i = 0; i < name_arr.length; i++)
        {
            if (name_arr[i] != arr[i])
                return false;
        }
        return true;
    } // check arr end
}
```

### 큰 수 만들기 (3번)

```java
class Solution {
    public String solution(String number, int k) {
        StringBuilder a = new StringBuilder(number);
        int i = 0;
        int j = 0;
        int idx;
        int len;

        for (i = 0; i < k; i++) {
            len = a.length();
            idx = len - 1;
            for (j = 0; j < len - 1; j++) {
                if (a.charAt(j) < a.charAt(j + 1)) {
                    idx = j;
                    break;
                }
            }
            a.deleteCharAt(idx);
        }

        return a.toString();
    }
}
```

```java
import java.util.Stack;
class Solution {
    public String solution(String number, int k) {
        char[] result = new char[number.length() - k];
        Stack<Character> stack = new Stack<>();

        for (int i=0; i<number.length(); i++) {
            char c = number.charAt(i);
            while (!stack.isEmpty() && stack.peek() < c && k-- > 0) {
                stack.pop();
            }
            stack.push(c);
        }
        for (int i=0; i<result.length; i++) {
            result[i] = stack.get(i);
        }
        return new String(result);
    }
}
```

### 구명보트 (4번)

```java
import java.util.*;
class Solution {
    public int solution(int[] people, int limit) {
        int answer = 0;
        Arrays.sort(people);
        boolean check[] = new boolean[people.length];
        int count = 0;
        int check_count = 0;
        int len = people.length;
        for (int i = len - 1; i > 0; i--)
        {
            if (people[0] + people[i] <= limit)
                break;
            check[i]  = true;
            count++;
            check_count++;
        }
        int temp = len - 1 - check_count;
        int a = 0;
        for (int i = 0; i < len - check_count; i++)
        {
            a = 0;
            if (check[i] == true)
                continue;
            if (i < len - 1 && people[i] + people[i + 1] <= limit)
            {
                for (int j = temp; j > i; j--)
                 {
                    if (check[j] == true)
                        continue;
                    if (people[i] + people[j] <= limit)
                    {
                        check[i] = true;
                        check[j] = true;
                        count++;
                        temp = j;
                        a = 1;
                        break;
                    }
                }
            }
            if (a == 0)
            {
                check[i] = true;
                count++;
            }
        }
        answer = count;
        return answer;
    }
}
```

```java
import java.util.Arrays;

class Solution {
    public int solution(int[] people, int limit) {
        Arrays.sort(people);
        int i = 0, j = people.length - 1;
        for (; i < j; --j) {
            if (people[i] + people[j] <= limit)
                ++i;
        }
        return people.length - i;
    }
}
```

## 깊이/너비 우선 탐색

### 타겟 넘버 (1번) 

```java
class Solution {
    static int count = 0;
    static int result = 0;
    static int len;
    static boolean[] visit;
    
    public int solution(int[] numbers, int target) {
        int answer = 0;
        len = numbers.length;
        visit = new boolean[len];
        dfs(numbers, 0, target);
        answer = count;
        return answer;
    } // solution end
    
    private void dfs(int[] numbers, int depth, int target)
    {
        if (depth == len)
        {
            if (result == target)
                count++;
            return;
        }
        if (visit[depth] == false)
        {
            int a = -1;
            for (int i = 0; i < 2; i++)
            {
                result += (numbers[depth] * a);
                visit[depth] = true;
                dfs(numbers, depth + 1, target);
                visit[depth] = false;
                result -= (numbers[depth] * a);
                a *= a;
            }
        }
        
    } // dfs end
}
```

```java
class Solution {
    public int solution(int[] numbers, int target) {
        int answer = 0;
        answer = dfs(numbers, 0, 0, target);
        return answer;
    }
    int dfs(int[] numbers, int n, int sum, int target) {
        if(n == numbers.length) {
            if(sum == target) {
                return 1;
            }
            return 0;
        }
        return dfs(numbers, n + 1, sum + numbers[n], target) + dfs(numbers, n + 1, sum - numbers[n], target);
    }
}
```

### 네트워크 (2번)

> 크루스칼 알고리즘을 비슷하게 사용하려고 했지만 변경사항이 있을 시 visit을 for 문으로 모두 확인하고 변경해야 하기 때문에.. (코드 확인)

```java
import java.util.*;
class Solution {
    public int solution(int n, int[][] computers) {
        int answer = 1;
        int visit[] = new int[n];
        for (int i = 0; i < n; i++)
            visit[i] = i;
        for (int i = 0; i < n; i++)
        {
            for (int j = 0; j < i; j++)
            {
                if (computers[i][j] == 1)
                    visit = check_com(visit, i, j, n);
            } // small for end
        } // big for end
        Arrays.sort(visit);
        for (int i = 0; i < n - 1; i++)
        {
           // System.out.println(visit[i]);
            if (visit[i] != visit[i + 1])
                answer++;
        }
       // System.out.println(visit[2]);
        return answer;
    } // solution end
    
    private int[] check_com(int[] visit, int f_n, int s_n, int n)
    {
        // i (f_n) > j (s_n)
        if (visit[f_n] < visit[s_n])
            visit[f_n] = visit[s_n];
        else
        {
            int temp = visit[s_n];
            for (int i = 0; i < f_n; i++)
            {
                if (visit[i] == temp)
                    visit[i] = visit[f_n];
            }
        }
        return visit;
    }
}
```

```java
class Solution {
    public int solution(int n, int[][] computers) {
        int answer = 0;
        boolean[] chk = new boolean[n];
        for(int i = 0; i < n; i++) {
            if(!chk[i]) {
                dfs(computers, chk, i);
                answer++;
            }
        }
        return answer;
    }
    void dfs(int[][] computers, boolean[] chk, int start) {
        chk[start] = true;
        for(int i = 0; i < computers.length; i++) {
            if(computers[start][i] == 1 && !chk[i]) {
                dfs(computers, chk, i);
            }
        }
    }
}
```

### 단어 변환 (3번)

```java
class Solution {
    static boolean t;
    static int count;
    static int temp_count;
    static boolean[] visit;
    
    public int solution(String begin, String target, String[] words) {
        int answer = 0;
        count = 100;
        temp_count = 0;
        visit = new boolean[words.length];
        bfs(begin, target, words, 0);
        if (t == true)
            return count;
        return answer;
    } // solution end
    
    private void bfs(String begin, String target, String[] words, int depth)
    {
        if (begin.equals(target))
        {
            t = true;
            count = Math.min(count, temp_count);
            return;
        }
        if (depth == words.length)
            return;
        String temp;
        for (int j = 0; j < words.length; j++)
        {
            if (visit[j] == false)
            {
                if (onlyOneDifferentChar(begin, words[j]))
                {
                    temp = begin;
                    begin = chage_char(begin, words[j]);
                    temp_count += 1;
                    visit[j] = true;
                    bfs(begin, target, words, depth + 1);
                    temp_count -= 1;
                    begin = temp;
                    visit[j] = false;
                } 
            }
        }  // big for end
        return;
    } // bfs end
    
    private String chage_char (String begin, String words)
    {
        String temp = "";
        for (int i = 0; i < begin.length(); i++)
        {
            if (begin.charAt(i) != words.charAt(i))
                temp += words.charAt(i);
            else
             temp += begin.charAt(i);
        }
        return temp;
    }
    
     private boolean onlyOneDifferentChar(String a, String b) {
        if (a.length() != b.length()) {
            return false;
        }

        int differentCnt = 0;
        for (int i = 0; i < a.length(); i++) {
            if (a.charAt(i) != b.charAt(i)) {
                differentCnt++;
            }
        }
        return differentCnt == 1;
     } // onlyOneDifferentChar end
}
```

```java
import java.util.LinkedList;
import java.util.Queue;

class Solution {

    static class Node {
        String next;
        int edge;

        public Node(String next, int edge) {
            this.next = next;
            this.edge = edge;
        }
    }

    public int solution(String begin, String target, String[] words) {
        int n = words.length, ans = 0;

        Queue<Node> q = new LinkedList<>();
        
        boolean[] visit = new boolean[n];
        q.add(new Node(begin, 0));

        while(!q.isEmpty()) {
            Node cur = q.poll();
            if (cur.next.equals(target)) {
                ans = cur.edge;
                break;
            }

            for (int i=0; i<n; i++) {
                if (!visit[i] && isNext(cur.next, words[i])) {
                    visit[i] = true;
                    q.add(new Node(words[i], cur.edge + 1));
                }
            }
        }

        return ans;
    }

    static boolean isNext(String cur, String n) {
        int cnt = 0;
        for (int i=0; i<n.length(); i++) {
            if (cur.charAt(i) != n.charAt(i)) {
                if (++ cnt > 1) return false;
            }
        }

        return true;
    }    
}
```

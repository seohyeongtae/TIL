```java
import java.util.*;
import java.io.*;

public class Main {
    static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
    static int[][] map;
    static int White = 0, Blue = 0;
    public static void main(String[] args) throws IOException{
     
        int N = Integer.parseInt(br.readLine());
        map = new int[N][N];

        for(int i = 0; i < N; i++){
            StringTokenizer st = new StringTokenizer(br.readLine());
            for(int j = 0; j < N; j++){
                map[i][j] = Integer.parseInt(st.nextToken());
            }
        }
        DAQ(map, 0, 0, N);

        System.out.println(White);
        System.out.println(Blue);
    }

    static void DAQ(int[][] arr, int y, int x, int S){
        
        int white = 0;
        int blue = 0;

        for(int i = y; i < y + S; i++){
            for(int j = x; j < x + S; j++){
                if(arr[i][j] == 1){
                    blue++;
                }else{
                    white++;
                }
            }
        }

        if(blue == 0){
            White++;
            return;
        }else if(white == 0){
            Blue++;
            return;
        }

        int nS = S / 2;

        DAQ(arr, y, x, nS);
        DAQ(arr, y, x + nS, nS);
        DAQ(arr, y + nS, x, nS);
        DAQ(arr, y + nS, x + nS, nS);
    }
}
```

```java
import java.util.*;
import java.io.*;
import java.math.*;

public class Main {

	public static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	public static BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
	public static String[][] arr;
	public static List<String> res = new ArrayList<>();

	public static void main(String[] args) throws IOException {
		String str = br.readLine();
		int n = Integer.parseInt(str);
		arr = new String[n][n];
		for (int i = 0; i < n; i++) {
			String[] line = br.readLine().split("");
			for (int j = 0; j < n; j++) {
				arr[i][j] = line[j];
			}
		}

		// 비교 함수 호출
		box(n, 0, 0);

		// 값 출력
		for (int i = 0; i < res.size(); i++) {
			System.out.print(res.get(i));
		}
		bw.flush();
		bw.close();
	}

	// 비교 함수
	public static void box(int total, int x, int y) {
		for (int i = x; i < x + total; i++) {
			for (int j = y; j < y + total; j++) {
				if (!arr[x][y].equals(arr[i][j])) { // 기준이 되는 값이 비교 값이랑 다른 경우
					// 4등분 된 구역마다 다시 비교 함수를 호출해준다.
					res.add("(");
					box(total / 2, x, y);
					box(total / 2, x, y + total / 2);
					box(total / 2, x + total / 2, y);
					box(total / 2, x + total / 2, y + total / 2);
					res.add(")");

					return;
				}
			}
		}
		res.add(arr[x][y]); //스트링에 저장
	}
}
```

    static void DAQ(int[][] arr, int y, int x, int S){
        
        int white = 0;
        int blue = 0;

        for(int i = y; i < y + S; i++){
            for(int j = x; j < x + S; j++){
                if(arr[i][j] == 1){
                    blue++;
                }else{
                    white++;
                }
            }
        }

        if(blue == 0){
            White++;
            return;
        }else if(white == 0){
            Blue++;
            return;
        }

        int nS = S / 2;

        DAQ(arr, y, x, nS);
        DAQ(arr, y, x + nS, nS);
        DAQ(arr, y + nS, x, nS);
        DAQ(arr, y + nS, x + nS, nS);

import java.util.*;
import java.io.*;
import java.math.*;

public class Main {

	public static BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
	public static BufferedWriter bw = new BufferedWriter(new OutputStreamWriter(System.out));
	public static String[][] arr;
	public static List<String> res = new ArrayList<>();

	public static void main(String[] args) throws IOException {
		String str = br.readLine();
		int n = Integer.parseInt(str);
		arr = new String[n][n];
		for (int i = 0; i < n; i++) {
			String[] line = br.readLine().split("");
			for (int j = 0; j < n; j++) {
				arr[i][j] = line[j];
			}
		}

		// 비교 함수 호출
		box(n, 0, 0);

		// 값 출력
		for (int i = 0; i < res.size(); i++) {
			System.out.print(res.get(i));
		}
		bw.flush();
		bw.close();
	}

	// 비교 함수
	public static void box(int total, int x, int y) {
		for (int i = x; i < x + total; i++) {
			for (int j = y; j < y + total; j++) {
				if (!arr[x][y].equals(arr[i][j])) { // 기준이 되는 값이 비교 값이랑 다른 경우
					// 4등분 된 구역마다 다시 비교 함수를 호출해준다.
					res.add("(");
					box(total / 2, x, y);
					box(total / 2, x, y + total / 2);
					box(total / 2, x + total / 2, y);
					box(total / 2, x + total / 2, y + total / 2);
					res.add(")");

					return;
				}
			}
		}
		res.add(arr[x][y]); //스트링에 저장
	}
}

import java.util.Scanner;
 
public class Main {
 
	public static long C;
	
	public static void main(String[] args) {
		
		Scanner in = new Scanner(System.in);
		
		long A = in.nextLong();
		long B = in.nextLong();
		C = in.nextLong();
		
		System.out.println(pow(A, B));
	}
	
	// A^exponent
	public static long pow(long A, long exponent) {
		
		// 지수가 1일 경우 A^1 이므로 A를 그대로 리턴
		if(exponent == 1) {
	//		System.out.println("final  "  + A % C);
			return A % C;
		}
		
		// 지수의 절반에 해당하는 A^(exponent / 2) 을 구한다.
		long temp = pow(A, exponent / 2);
		//System.out.println("temp ===  " + temp );
		
		/*
		 * 현재 지수가 홀 수 였다면
		 * A^(exponent / 2) * A^(exponent / 2) * A 이므로
		 * A를 한 번 더 곱해주어야 한다.
		 * 
		 * ex) A^9 = A^4 * A^4 * A
		 */
		if(exponent % 2 == 1) {
	//		System.out.println("how  "  + (temp * temp % C) * A % C);
			return (temp * temp % C) * A % C;
		}
	//	System.out.println("second  " + temp * temp % C );
		return temp * temp % C;
		
	}
	
}

### 이항계수 3 (11401번)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;


public class Main {
	
	static long P = 1000000007;
	public static void main(String[] args) throws IOException {
		
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		long a = Integer.parseInt(st.nextToken());
		long b = Integer.parseInt(st.nextToken());


		// N!  (1)
		long first = factorial(a);
		// K! * N-K! (2)
		long second = factorial(b) * factorial(a-b) % P;

		// 1 * 2 % P
		System.out.println(first * pow(second, P - 2) % P);
		

	} // main end
	private static long factorial(long a) {
		long temp = 1l;
		while (a > 1)
		{
			temp = (temp * a) % P;
			a--;
		}
		return temp;
	}
	private static long pow(long a, long b) {

		if (b == 1)
			return a % P;

		long temp = pow(a, b / 2);

		if (b % 2 == 1)
		{
			// temp = temp * temp * a % P
			return (temp * temp % P) * a % P;
		}
		else
		{
			return temp * temp % P;
		}
	}
} // class end
```

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;


public class Main {
	
	static StringBuilder sb = new StringBuilder();
	static int first[][];
	static int second[][];
	public static void main(String[] args) throws IOException {
		
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int a = Integer.parseInt(st.nextToken());
		int b = Integer.parseInt(st.nextToken());

		first = new int[a][b];
		for (int i = 0; i < a; i++)
		{
			st = new StringTokenizer(br.readLine());
			for (int j = 0; j < b; j++)
			{
				int a1 = Integer.parseInt(st.nextToken());
				first[i][j] = a1;
			}
		}
		st = new StringTokenizer(br.readLine());
		// aa == b
		int aa = Integer.parseInt(st.nextToken());
		int bb = Integer.parseInt(st.nextToken());
		//System.out.println(aa  + " aa  " + bb);
		second = new int[aa][bb];
		for (int i = 0; i < aa; i++)
		{
			st = new StringTokenizer(br.readLine());
			for (int j = 0; j < bb; j++)
			{
				int a1 = Integer.parseInt(st.nextToken());
				second[i][j] = a1;
			}
		}


		for (int i = 0; i < a; i++)
		{
			for (int j = 0; j < bb; j++)
			{
				make_result(i, b, j);
			}
			sb.append('\n');
		}
		System.out.println(sb);
	} // main end
	private static void make_result(int i, int b, int bb) {
		int result = 0;
		
		for (int j = 0; j < b; j++)
		{
			result += first[i][j] * second[j][bb];
		}
		sb.append(result + " ");
	}

} // class end

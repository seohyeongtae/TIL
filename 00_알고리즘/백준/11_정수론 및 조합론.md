### 최대공약수와 최소공배수 (2609번)

> 유클리드 호제법
>
> 두 정수 *a*, *b*의 최대공약수를 *G*(*a*, *b*)라고 하자.
>
> 정수 *a*, *b*, *q* *r* (*b* ≠ 0)에 대하여 *a* = *bq* + *r*,이면 *G*(*a*, *b*) = *G*(*b*, *r*)가 성립한다.
>
> [링크1](https://terms.naver.com/entry.naver?docId=2073670&cid=47324&categoryId=47324) [링크2](https://blog.naver.com/galaxyenergy/222264614390) [링크3](https://blog.naver.com/gdpresent/222452479379)

### 검문 (2981번)

> (M : 나눈 수 r : 나머지(remainder))
>
>***A1 = M \* a1 + r1\***  
>
>***A2 = M \* a2 + r2\***
>
>이 때 나머지가 같아야 하기 때문에 r1과 r2가 같아야 한다는 가정하에 A1에서 A2를 빼면 다음과 같이 만들 수 있다.
>
>***A1 - A2 = M \* (a1 - a2) + (r1 - r2)\***
>
>*r1 - r2 = 0* 이므로, ***A1 - A2 = M \* (a1 - a2)\*** 
>
>*M* 은 *(A1 - A2)* 의 약수이므로, *A1* 과 *A2* 의 공약수가 됨.
>
>즉, 우리는 여기서 M을 찾아야 한다

```java
package baekjoon;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;
import java.io.IOException;
 
public class Main {
	
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		int n = Integer.parseInt(st.nextToken());
		int arr[] = new int[n];
		int arr2[] = new int[n - 1];
		for (int i = 0; i < n; i++)
		{
			st = new StringTokenizer(br.readLine());
			arr[i] = Integer.parseInt(st.nextToken());
		}
		Arrays.sort(arr);
		for (int i = 0; i < n - 1; i++)
			arr2[i] = arr[i + 1] - arr[i];
		Arrays.sort(arr2);
		for (int i = 0 ; i < n - 2; i++)
		{
			while (arr2[i] != 0)
			{
				int r = arr2[i + 1] % arr2[i];
				arr2[i + 1] = arr2[i];
				arr2[i] = r;
			}
		}
		for (int i = 2; i <= arr2[n - 2]; i++)
		{
			if (arr2[n - 2] % i == 0)
				System.out.print(i + " ");
		}
	} // main end
} // class end
```

```java
		}
 
 
		// 최대공약수의 약수들 찾기
		for(int i = 2; i <= gcdVal; i++) {
	    
			// i가 최대공약수의 약수라면 출력

				System.out.print(i + " ");
			}

	static int gcd(int a, int b) {
		while(b != 0) {
			int r = a % b;
			a = b;
			b = r;
		}
		return a;
	}
```

### 이항계수 2 (11051번)

>* [모듈라연산](https://st-lab.tistory.com/162)   - ex>  **(A+B) %C = (A%C + B%C)%C**   / **(A\*B)%C = (A%C \* B%C)%C** 
>
>  > 모듈라 연산에서 나눗셈 연산은 없다!  **(A%B) %C != ((A%C) % (B%C)%C**
>
>  * ***a mod m = r\*** 표현은 **a % m = r** 이랑 같은 의미
>  * (34 mod 7) = (27 mod 7) = (62 mod 7) = 6  → (34 % 7) == (27 % 7) == (62 % 7) == 6
>  * 반대로 생각하면 ***a = km + r*** (나눈 수 m의 k배수에 r을 더한 값은 a를 만족한다) (k = a div m (몫)이랑 같다.)
>  * (34 mod 7 = 6) → (34 = 4 × 7 + 6)           (27 mod 7 = 6) → (27 = 4 × 3 + 6)
>
>* [페르마의 소정리](https://ko.wikipedia.org/wiki/%ED%8E%98%EB%A5%B4%EB%A7%88%EC%9D%98_%EC%86%8C%EC%A0%95%EB%A6%AC)
>

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;
 
public class Main {
```
```java
	public static final int div = 10007;
 
	public static void main(String[] args) throws IOException {
 
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
 
		StringTokenizer st = new StringTokenizer(br.readLine(), " ");
 
		int N = Integer.parseInt(st.nextToken());
		int K = Integer.parseInt(st.nextToken());
 
		/*
		 *   n! / ((n-k)! * k!)   ->   n! * ((n-k)! * k!)^(-1) 으로 변환
		 *   ((n-k)! * k!)^(-1) == ((n-k)! * k!)^(p-2) 동치  
		 *   p(=div)가 소수여서 가능함)
		 */
		System.out.println((factorial(N) * mod_inverse((factorial(N - K) * factorial(K)) % div, div - 2)) % div);
	}
 
	static int factorial(int N) {
		// factorial(0) == 1 이다.
		if (N <= 1) {
			return 1;
		}
		return (N * factorial(N - 1)) % div;
	}
 
	// 역원 구하기 (= 제곱승 구하기)
	static int mod_inverse(int a, int p) {
		int ret = 1;
		while(p > 0) {
			if(p % 2 == 1) {
				ret *= a;
				p--;
				ret %= div;
			}
			a *= a;
			a %= div;
			p >>= 1;	// p = p/2 랑 동치 
		}
		return ret;
	}
}
```
